> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Jogo jogável na cena `Jogo.unity`: F1 com física dirigindo num terreno infinito de Bézier (recortado por Sutherland–Hodgman e preenchido por scanline), câmera que segue, combustível, moedas, HUD, pausa, fim de jogo e reinício.

## Review Focus

1. **Emenda entre trechos** — a roda não pode travar/quicar na junção de dois `EdgeCollider2D`. Teste: `JogoTests.Acelerando_AtravessaVariosTrechos` (Task 9) exige avançar > 40 unidades, cruzando ≥ 2 emendas.
2. **Sessão longa** — trechos antigos (GameObject, textura, sprite) precisam ser liberados. Teste: `TerrenoComponenteTests.Atualizar_DescartaTrechosAntigos` (Task 3).
3. **Reiniciar com o jogo pausado** — `Time.timeScale` não pode ficar 0 na cena nova. Teste: `JogoTests.Pausar_ReiniciarRestauraTempo` (Task 9).
4. **Terreno subindo/descendo sem limite** — altura limitada por `limiteY`. Teste: `TerrenoTests.Trechos_XCrescenteEAlturaLimitada` (Task 2).
5. **Morro impossível de subir / curva voltando em x** — inclinação ≤ 1,5 e controles com x crescente. Testes: `TerrenoTests.Trechos_InclinacaoJogavel` e `Trechos_XCrescenteEAlturaLimitada` (Task 2).

`Tools/extrair_unitypackage.py`:
```python
"""Extrai um .unitypackage (tar.gz de pastas <guid>/{asset, asset.meta, pathname}) dentro de <projeto>/Assets.
Uso: python3 -I Tools/extrair_unitypackage.py <pacote.unitypackage> <pasta-do-projeto>"""
import os, sys, tarfile

pacote, projeto = sys.argv[1], os.path.abspath(sys.argv[2])
raiz = os.path.join(projeto, 'Assets') + os.sep
with tarfile.open(pacote, 'r:gz') as t:
    entradas = {}
    for m in t.getmembers():
        partes = m.name.lstrip('./').split('/')
        if len(partes) == 2 and partes[1]:
            entradas.setdefault(partes[0], {})[partes[1]] = m
    for arquivos in entradas.values():
        if 'pathname' not in arquivos:
            continue
        caminho = t.extractfile(arquivos['pathname']).read().decode('utf-8').splitlines()[0].strip()
        destino = os.path.abspath(os.path.join(projeto, caminho))
        if not destino.startswith(raiz):
            raise SystemExit('caminho fora de Assets: ' + caminho)
        if 'asset' in arquivos:
            os.makedirs(os.path.dirname(destino), exist_ok=True)
            with open(destino, 'wb') as f:
                f.write(t.extractfile(arquivos['asset']).read())
        else:
            os.makedirs(destino, exist_ok=True)
        if 'asset.meta' in arquivos:
            with open(destino + '.meta', 'wb') as f:
                f.write(t.extractfile(arquivos['asset.meta']).read())
        print(caminho)
```

```csharp
    // Emenda C1 entre trechos: o 1º controle do novo trecho é o espelhamento do último controle
    // do anterior em torno do ponto de junção.
    public static Vector2 BezierJoin(Vector2 p2Anterior, Vector2 p3Anterior) => 2 * p3Anterior - p2Anterior;

    // Recorte de polígono de Sutherland–Hodgman: recorta contra cada borda do retângulo, em sequência.
    public static List<Vector2> ClipPolygon(IReadOnlyList<Vector2> poly, Rect r)
    {
        var p = new List<Vector2>(poly);
        p = ClipBorda(p, v => v.x >= r.xMin, (a, b) => Corte(a, b, (r.xMin - a.x) / (b.x - a.x)));
        p = ClipBorda(p, v => v.x <= r.xMax, (a, b) => Corte(a, b, (r.xMax - a.x) / (b.x - a.x)));
        p = ClipBorda(p, v => v.y >= r.yMin, (a, b) => Corte(a, b, (r.yMin - a.y) / (b.y - a.y)));
        p = ClipBorda(p, v => v.y <= r.yMax, (a, b) => Corte(a, b, (r.yMax - a.y) / (b.y - a.y)));
        return p;
    }

    static Vector2 Corte(Vector2 a, Vector2 b, float t) => a + (b - a) * t;

    // Uma borda: para cada aresta (anterior → atual), emite a interseção ao cruzar a borda e o vértice se estiver dentro.
    static List<Vector2> ClipBorda(List<Vector2> entrada, System.Func<Vector2, bool> dentro,
                                   System.Func<Vector2, Vector2, Vector2> intersecao)
    {
        var saida = new List<Vector2>();
        for (int i = 0; i < entrada.Count; i++)
        {
            Vector2 atual = entrada[i], anterior = entrada[(i + entrada.Count - 1) % entrada.Count];
            bool dAtual = dentro(atual), dAnterior = dentro(anterior);
            if (dAtual)
            {
                if (!dAnterior) saida.Add(intersecao(anterior, atual));
                saida.Add(atual);
            }
            else if (dAnterior) saida.Add(intersecao(anterior, atual));
        }
        return saida;
    }
```

```csharp
using System.Linq;
using UnityEngine;

public struct Trecho { public Vector2 p0, p1, p2, p3; }

// Sequência infinita de Bézier cúbicas com emenda C1. Morros crescem com a distância.
public class TerrenoGerador
{
    public float largura = 20f;
    public float amplitudeInicial = 1.5f;
    public float ganhoPorTrecho = 0.25f;
    public float amplitudeMax = 5f;
    public float limiteY = 30f;

    readonly System.Random rnd;
    Trecho? anterior;

    public int Indice { get; private set; }

    public TerrenoGerador(int semente) => rnd = new System.Random(semente);

    float Sorteio(float a, float b) => a + (float)rnd.NextDouble() * (b - a);

    public Trecho Proximo()
    {
        Trecho t;
        if (anterior == null)   // largada plana
        {
            t = new Trecho { p0 = new(0, 0), p1 = new(largura / 3, 0), p2 = new(2 * largura / 3, 0), p3 = new(largura, 0) };
        }
        else
        {
            var a = anterior.Value;
            float amp = Mathf.Min(amplitudeMax, amplitudeInicial + ganhoPorTrecho * Indice);
            float x0 = a.p3.x;
            float y3 = Mathf.Clamp(a.p3.y + Sorteio(-amp, amp), -limiteY, limiteY);
            t = new Trecho
            {
                p0 = a.p3,
                p1 = Raster.BezierJoin(a.p2, a.p3),                              // espelhamento → C1
                p2 = new(x0 + largura * Sorteio(0.6f, 0.75f), y3 + Sorteio(-0.3f, 0.3f) * amp),
                p3 = new(x0 + largura, y3),
            };
        }
        anterior = t;
        Indice++;
        return t;
    }

    public static Vector2[] Amostrar(Trecho t, int n) =>
        Enumerable.Range(0, n).Select(i => Raster.BezierPoint(t.p0, t.p1, t.p2, t.p3, i / (float)(n - 1))).ToArray();

    // Normal unitária à esquerda da tangente: "para cima" quando a curva avança em +x.
    public static Vector2 Normal(Trecho t, float u)
    {
        var d = Raster.BezierTangent(t.p0, t.p1, t.p2, t.p3, u).normalized;
        return new Vector2(-d.y, d.x);
    }
}
```

```csharp
using System.Collections.Generic;
using System.Linq;
using UnityEngine;

// Parte pura de um trecho: polígono, altura da superfície e preenchimento por scanline num buffer de cores.
public static class TerrenoChunk
{
    [System.Serializable]
    public class Paleta
    {
        public Color gramaBorda = new(0.62f, 0.86f, 0.36f);
        public Color grama = new(0.36f, 0.66f, 0.24f);
        public Color terraClara = new(0.55f, 0.38f, 0.24f);
        public Color terraEscura = new(0.28f, 0.18f, 0.12f);
        public float espessuraBorda = 0.07f, espessuraGrama = 0.35f, profundidadeGradiente = 8f;

        // Cor pela profundidade abaixo da superfície (em unidades).
        public Color32 Cor(float profundidade) =>
            profundidade < espessuraBorda ? gramaBorda :
            profundidade < espessuraGrama ? grama :
            Color.Lerp(terraClara, terraEscura, Mathf.Clamp01((profundidade - espessuraGrama) / profundidadeGradiente));
    }

    // Curva da superfície + dois cantos de baixo.
    public static List<Vector2> Poligono(Vector2[] curva, float fundoY)
    {
        var p = new List<Vector2>(curva);
        p.Add(new Vector2(curva[^1].x, fundoY));
        p.Add(new Vector2(curva[0].x, fundoY));
        return p;
    }

    // Altura da superfície em x, por interpolação linear entre as amostras (curva com x crescente).
    public static float AlturaEm(Vector2[] curva, float x)
    {
        if (x <= curva[0].x) return curva[0].y;
        for (int i = 1; i < curva.Length; i++)
            if (x <= curva[i].x)
                return Mathf.Lerp(curva[i - 1].y, curva[i].y, (x - curva[i - 1].x) / (curva[i].x - curva[i - 1].x));
        return curva[^1].y;
    }

    // Preenche o polígono (em coordenadas de mundo) por scanline numa grade de ppu pixels por unidade
    // cobrindo a janela. Pixel fora do polígono fica transparente.
    public static Color32[] Pintar(IReadOnlyList<Vector2> poligono, Rect janela, float ppu, Vector2[] curva,
                                   Paleta paleta, out int w, out int h)
    {
        w = Mathf.CeilToInt(janela.width * ppu);
        h = Mathf.CeilToInt(janela.height * ppu);
        var px = new Color32[w * h];
        if (poligono.Count < 3) return px;

        var superficie = new float[w];
        for (int x = 0; x < w; x++) superficie[x] = AlturaEm(curva, janela.xMin + (x + 0.5f) / ppu);

        var pix = poligono.Select(p => Vector2Int.RoundToInt((p - janela.min) * ppu)).ToArray();
        foreach (var (y, x0, x1) in Raster.ScanlineSpans(pix))
        {
            if (y < 0 || y >= h) continue;
            float yMundo = janela.yMin + (y + 0.5f) / ppu;
            for (int x = Mathf.Max(0, x0); x <= Mathf.Min(w - 1, x1); x++)
                px[y * w + x] = paleta.Cor(superficie[x] - yMundo);
        }
        return px;
    }
}
```

```csharp
using System.Collections.Generic;
using System.Linq;
using UnityEngine;

// Terreno infinito: trecho de Bézier → polígono recortado (Sutherland–Hodgman) → textura por scanline + collider.
public class Terreno : MonoBehaviour
{
    public class Chunk
    {
        public Trecho trecho;
        public int indice;
        public Vector2[] curva;
        public Rect janela;
        public GameObject go;
        public Texture2D tex;
    }

    public Transform alvo;
    public int semente = 7;
    public float ppu = 64;
    public int amostras = 64;
    public int trechosAFrente = 3;
    public float profundidade = 12f;
    public float alturaMaxJanela = 32f;
    public TerrenoChunk.Paleta paleta = new();
    public PhysicsMaterial2D materialChao;

    public readonly List<Chunk> chunks = new();
    public event System.Action<Chunk> TrechoCriado;
    public TerrenoGerador Gerador { get; private set; }

    void Awake() => Reiniciar();

    void Update()
    {
        if (alvo) Atualizar(alvo.position.x);
    }

    public void Reiniciar()
    {
        foreach (var c in chunks) Liberar(c);
        chunks.Clear();
        Gerador = new TerrenoGerador(semente);
    }

    // Mantém trechos até trechosAFrente larguras à frente de x e descarta os que ficaram uma largura para trás.
    public void Atualizar(float x)
    {
        while (chunks.Count == 0 || chunks[^1].trecho.p3.x < x + trechosAFrente * Gerador.largura) GerarProximo();
        while (chunks.Count > 1 && chunks[0].trecho.p3.x < x - Gerador.largura)
        {
            Liberar(chunks[0]);
            chunks.RemoveAt(0);
        }
    }

    public Chunk GerarProximo()
    {
        var t = Gerador.Proximo();
        int indice = Gerador.Indice - 1;
        var curva = TerrenoGerador.Amostrar(t, amostras);
        float topo = curva.Max(p => p.y) + 0.5f;
        float fundo = curva.Min(p => p.y) - profundidade;
        var janela = Rect.MinMaxRect(t.p0.x, Mathf.Max(fundo, topo - alturaMaxJanela), t.p3.x, topo);

        var poligono = Raster.ClipPolygon(TerrenoChunk.Poligono(curva, fundo), janela);          // recorte
        var px = TerrenoChunk.Pintar(poligono, janela, ppu, curva, paleta, out int w, out int h);   // preenchimento

        var tex = new Texture2D(w, h, TextureFormat.RGBA32, false)
            { wrapMode = TextureWrapMode.Clamp, filterMode = FilterMode.Bilinear, name = $"Trecho {indice}" };
        tex.SetPixels32(px);
        tex.Apply();

        var go = new GameObject($"Trecho {indice}");
        go.transform.SetParent(transform, false);
        go.transform.position = janela.min;
        go.AddComponent<SpriteRenderer>().sprite = Sprite.Create(tex, new Rect(0, 0, w, h), Vector2.zero, ppu);
        var col = go.AddComponent<EdgeCollider2D>();
        col.points = curva.Select(p => p - janela.min).ToArray();
        col.sharedMaterial = materialChao;
        if (chunks.Count > 0) Emendar(chunks[^1], go, col);

        var c = new Chunk { trecho = t, indice = indice, curva = curva, janela = janela, go = go, tex = tex };
        chunks.Add(c);
        TrechoCriado?.Invoke(c);
        return c;
    }

    // Pontos vizinhos nas emendas: a roda passa de um collider para o outro sem "degrau".
    static void Emendar(Chunk anterior, GameObject go, EdgeCollider2D col)
    {
        var colAnt = anterior.go.GetComponent<EdgeCollider2D>();
        Vector2 o = go.transform.position, oAnt = anterior.go.transform.position;
        colAnt.useAdjacentEndPoint = true;
        colAnt.adjacentEndPoint = col.points[1] + o - oAnt;
        col.useAdjacentStartPoint = true;
        col.adjacentStartPoint = colAnt.points[^2] + oAnt - o;
    }

    static void Liberar(Chunk c)
    {
        Destruir(c.go.GetComponent<SpriteRenderer>().sprite);
        Destruir(c.go);
        Destruir(c.tex);
    }

    static void Destruir(Object o)
    {
        if (Application.isPlaying) Destroy(o);
        else DestroyImmediate(o);
    }
}
```

```csharp
    public float consumoAcelerando = 2.5f;
    public float tempoParadoParaPane = 2f;
    public float velocidadeParado = 0.3f;

    float parado;

    public float Atual { get; private set; }
    public bool Vazio => Atual <= 0;
    public float Fracao => Atual / maximo;

    public void Encher()
    {
        Atual = maximo;
        parado = 0;
    }

    public void Consumir(float dt, float acelerador) =>
        Atual = Mathf.Max(0, Atual - dt * (consumoPorSegundo + consumoAcelerando * Mathf.Abs(acelerador)));

    public void Abastecer(float qtd) => Atual = Mathf.Min(maximo, Atual + qtd);

    // Fim de jogo por pane seca: sem combustível e parado por tempoParadoParaPane segundos seguidos.
    public bool VerificarPane(float dt, float velocidade)
    {
        parado = Vazio && Mathf.Abs(velocidade) < velocidadeParado ? parado + dt : 0;
        return parado >= tempoParadoParaPane;
    }
}
```

```csharp
using System.Collections.Generic;
using UnityEngine;

public enum TipoColetavel { Moeda, Galao }

// Onde ficam moedas e galões em cada trecho (puro). Posição = ponto da curva + normal · altura (translação).
public class PlanoColetaveis
{
    int proximoGalao = 2, intervalo = 3;

    public List<(Vector2 pos, TipoColetavel tipo)> Para(Trecho t, int indice)
    {
        var lista = new List<(Vector2, TipoColetavel)>();
        if (indice == 0) return lista;
        for (int k = 0; k < 5; k++) lista.Add((Acima(t, 0.3f + 0.05f * k, 1f), TipoColetavel.Moeda));
        if (indice == proximoGalao)
        {
            lista.Add((Acima(t, 0.8f, 1.2f), TipoColetavel.Galao));
            proximoGalao += intervalo;
            intervalo++;
        }
        return lista;
    }

    static Vector2 Acima(Trecho t, float u, float altura) =>
        Raster.BezierPoint(t.p0, t.p1, t.p2, t.p3, u) + TerrenoGerador.Normal(t, u) * altura;
}

public class Coletavel : MonoBehaviour
{
    public TipoColetavel tipo;
    public static event System.Action<TipoColetavel> Coletado;
    bool pego;

    void Update()
    {
        if (tipo != TipoColetavel.Moeda) return;
        var s = transform.localScale;               // moeda girando: escala x = cos(θ)
        s.x = Mathf.Cos(Time.time * 4f + transform.position.x);
        transform.localScale = s;
    }

    void OnTriggerEnter2D(Collider2D outro)
    {
        if (pego || !outro.CompareTag("Player")) return;   // roda e chassi podem tocar no mesmo passo
        pego = true;
        Coletado?.Invoke(tipo);
        Destroy(gameObject);
    }
}

// Cria os coletáveis de cada trecho novo, como filhos do trecho (somem junto com ele).
public class Coletaveis : MonoBehaviour
{
    public Terreno terreno;
    public Sprite moeda, galao;
    public float raio = 0.3f;

    readonly PlanoColetaveis plano = new();

    void OnEnable() => terreno.TrechoCriado += Popular;
    void OnDisable() => terreno.TrechoCriado -= Popular;

    void Popular(Terreno.Chunk c)
    {
        foreach (var (pos, tipo) in plano.Para(c.trecho, c.indice))
        {
            var go = new GameObject(tipo.ToString());
            go.transform.SetParent(c.go.transform, false);
            go.transform.position = pos;
            var sr = go.AddComponent<SpriteRenderer>();
            sr.sprite = tipo == TipoColetavel.Moeda ? moeda : galao;
            sr.sortingOrder = 1;
            var col = go.AddComponent<CircleCollider2D>();
            col.isTrigger = true;
            col.radius = raio;
            go.AddComponent<Coletavel>().tipo = tipo;
        }
    }
}
```

```csharp
// F1 com tração traseira: motor na roda traseira, torque no chassi quando está no ar, sensor no capacete.
[RequireComponent(typeof(Rigidbody2D))]
public class F1 : MonoBehaviour
{
    public WheelJoint2D eixoTraseiro, eixoDianteiro;
    public float velocidadeMotor = 2400f;      // graus/s na roda
    public float torqueMotor = 40f;
    public float torqueNoAr = 8f;
    public Vector2 centroDeMassa = new(0, -0.15f);
    public bool controlePeloTeclado = true;
    [Range(-1, 1)] public float acelerador;
    public bool motorLigado = true;

    public event System.Action Capotou;
    public bool NoChao { get; private set; }
    public float VelocidadeX => rb.linearVelocity.x;

    Rigidbody2D rb;
    Collider2D rodaT, rodaD;

    void Awake()
    {
        rb = GetComponent<Rigidbody2D>();
        rb.centerOfMass = centroDeMassa;
        rodaT = eixoTraseiro.connectedBody.GetComponent<Collider2D>();
        rodaD = eixoDianteiro.connectedBody.GetComponent<Collider2D>();
    }

    void Update()
    {
        if (controlePeloTeclado)
            acelerador = (Input.GetKey(KeyCode.RightArrow) ? 1 : 0) - (Input.GetKey(KeyCode.LeftArrow) ? 1 : 0);
    }

    void FixedUpdate()
    {
        NoChao = rodaT.IsTouchingLayers() || rodaD.IsTouchingLayers();
        float a = motorLigado ? acelerador : 0;
        eixoTraseiro.useMotor = a != 0;                                    // sem acelerar: roda livre
        eixoTraseiro.motor = new JointMotor2D { motorSpeed = -a * velocidadeMotor, maxMotorTorque = torqueMotor };
        if (!NoChao) rb.AddTorque(acelerador * torqueNoAr);                // → levanta o bico, ← abaixa (rotação)
    }

    public void NotificarCapotou() => Capotou?.Invoke();
}

public class SensorCapacete : MonoBehaviour
{
    public F1 f1;

    void OnTriggerEnter2D(Collider2D outro)
    {
        if (outro is EdgeCollider2D) f1.NotificarCapotou();
    }
}
```

```csharp
using UnityEngine;

// Segue o carro olhando à frente na direção da velocidade e abre o zoom com a velocidade.
[RequireComponent(typeof(Camera))]
public class CameraSegue : MonoBehaviour
{
    public Rigidbody2D alvo;
    public Vector2 deslocamento = new(3f, 1.5f);
    public float suavidade = 0.25f;
    public float olharAFrente = 0.3f, olharMax = 4f;
    public float zoomMin = 7f, zoomMax = 11f, velocidadeZoomMax = 18f, suavidadeZoom = 0.8f;

    Camera cam;
    Vector3 vel;
    float velZoom;

    void Awake() => cam = GetComponent<Camera>();

    void LateUpdate()
    {
        Vector2 v = alvo.linearVelocity;
        Vector2 frente = Vector2.ClampMagnitude(v * olharAFrente, olharMax);
        Vector3 destino = alvo.position + deslocamento + frente;
        destino.z = transform.position.z;
        transform.position = Vector3.SmoothDamp(transform.position, destino, ref vel, suavidade);
        float zoom = Mathf.Lerp(zoomMin, zoomMax, Mathf.Clamp01(v.magnitude / velocidadeZoomMax));
        cam.orthographicSize = Mathf.SmoothDamp(cam.orthographicSize, zoom, ref velZoom, suavidadeZoom);
    }
}
```

```csharp
    public Estado EstadoAtual { get; private set; }
    public int Moedas { get; private set; }
    public float Distancia { get; private set; }

    const string ChaveRecorde = "recorde";
    float x0;

    void OnEnable()
    {
        Coletavel.Coletado += AoColetar;
        f1.Capotou += AoCapotar;
    }

    void OnDisable()
    {
        Coletavel.Coletado -= AoColetar;
        f1.Capotou -= AoCapotar;
        Time.timeScale = 1;
    }

    void Start()
    {
        Time.timeScale = 1;
        combustivel.Encher();
        x0 = f1.transform.position.x;
        painelPausa.SetActive(false);
        painelFim.SetActive(false);
        AtualizarHud();
    }

    void Update()
    {
        if (Input.GetKeyDown(KeyCode.R)) { Reiniciar(); return; }
        if (Input.GetKeyDown(KeyCode.Escape) && EstadoAtual != Estado.Fim) AlternarPausa();
        if (EstadoAtual != Estado.Jogando) return;

        combustivel.Consumir(Time.deltaTime, f1.motorLigado ? f1.acelerador : 0);
        f1.motorLigado = !combustivel.Vazio;
        Distancia = Mathf.Max(Distancia, f1.transform.position.x - x0);
        if (combustivel.VerificarPane(Time.deltaTime, f1.VelocidadeX)) Terminar("Sem combustível");
        AtualizarHud();
    }

    public void Reiniciar() => SceneManager.LoadScene(SceneManager.GetActiveScene().buildIndex);

    public void AlternarPausa()
    {
        bool pausar = EstadoAtual == Estado.Jogando;
        EstadoAtual = pausar ? Estado.Pausa : Estado.Jogando;
        Time.timeScale = pausar ? 0 : 1;
        painelPausa.SetActive(pausar);
    }

    void AoColetar(TipoColetavel tipo)
    {
        if (tipo == TipoColetavel.Moeda) Moedas++;
        else combustivel.Abastecer(galaoQuantidade);
    }

    void AoCapotar()
    {
        if (EstadoAtual == Estado.Jogando) Terminar("Capotou!");
    }

    public void Terminar(string motivo)
    {
        EstadoAtual = Estado.Fim;
        f1.motorLigado = false;
        float recorde = PlayerPrefs.GetFloat(ChaveRecorde, 0);
        if (Distancia > recorde)
        {
            recorde = Distancia;
            PlayerPrefs.SetFloat(ChaveRecorde, recorde);
            PlayerPrefs.Save();
        }
        textoFim.text = $"{motivo}\n<size=60%>{Distancia:0} m  ·  {Moedas} moedas\nRecorde: {recorde:0} m\n\nR para jogar de novo</size>";
        painelFim.SetActive(true);
        AtualizarHud();
    }

    void AtualizarHud()
    {
        var a = barraCombustivel.anchorMax;
        a.x = combustivel.Fracao;
        barraCombustivel.anchorMax = a;
        textoMoedas.text = $"Moedas  {Moedas}";
        textoDistancia.text = $"{Distancia:0} m";
        textoRecorde.text = $"Recorde {PlayerPrefs.GetFloat(ChaveRecorde, 0):0} m";
    }
}
```
